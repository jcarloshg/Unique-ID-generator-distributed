# Phase 202 — Hexagonal Architecture

## Overview

## Domain Layer

### SnowflakeId (Value Object)

```python
# src/generator/domain/models/snowflake_id.py
from dataclasses import dataclass

EPOCH = 1288834974657

@dataclass(frozen=True)
class SnowflakeId:
    """Immutable 64-bit Snowflake ID value object."""
    value: int

    @property
    def timestamp_ms(self) -> int:
        return (self.value >> 22) + EPOCH

    @property
    def datacenter_id(self) -> int:
        return (self.value >> 17) & 0x1F

    @property
    def machine_id(self) -> int:
        return (self.value >> 12) & 0x1F

    @property
    def sequence(self) -> int:
        return self.value & 0xFFF

    @classmethod
    def from_components(
        cls,
        timestamp_ms: int,
        datacenter_id: int,
        machine_id: int,
        sequence: int
    ) -> 'SnowflakeId':
        value = (
            (timestamp_ms << 22) |
            (datacenter_id << 17) |
            (machine_id << 12) |
            sequence
        )
        return cls(value=value)

    def to_dict(self) -> dict:
        return {
            'id': self.value,
            'timestamp_ms': self.timestamp_ms,
            'datacenter_id': self.datacenter_id,
            'machine_id': self.machine_id,
            'sequence': self.sequence,
        }
```

### IdGeneratorService (Domain Service)

```python
# src/generator/domain/services/id_generator.py
from typing import Protocol
import asyncio

from ..models.snowflake_id import SnowflakeId
from ..ports.clock import IClock

class IdGeneratorService:
    """Pure domain service for generating Snowflake IDs."""

    def __init__(
        self,
        datacenter_id: int,
        machine_id: int,
        clock: IClock,
    ):
        if not 0 <= datacenter_id <= 31:
            raise ValueError("datacenter_id must be 0-31")
        if not 0 <= machine_id <= 31:
            raise ValueError("machine_id must be 0-31")

        self._datacenter_id = datacenter_id
        self._machine_id = machine_id
        self._clock = clock
        self._sequence = 0
        self._last_timestamp_ms = 0
        self._lock = asyncio.Lock()

    async def generate_id(self) -> SnowflakeId:
        async with self._lock:
            current_ms = self._clock.get_current_time_ms()

            # Handle same millisecond
            if current_ms == self._last_timestamp_ms:
                self._sequence += 1
                if self._sequence >= 4096:
                    while current_ms == self._last_timestamp_ms:
                        current_ms = self._clock.get_current_time_ms()
                        await asyncio.sleep(0.001)
                    self._sequence = 0
            else:
                self._sequence = 0

            # Clock validation
            if current_ms < self._last_timestamp_ms:
                drift = self._last_timestamp_ms - current_ms
                if drift > 10:
                    raise ClockBackwardsError(f"Clock backwards by {drift}ms")

            self._last_timestamp_ms = current_ms

            return SnowflakeId.from_components(
                timestamp_ms=current_ms,
                datacenter_id=self._datacenter_id,
                machine_id=self._machine_id,
                sequence=self._sequence,
            )
```

## Ports (Interfaces)

### IClock

```python
# src/generator/domain/ports/clock.py
from abc import ABC, abstractmethod

class IClock(ABC):
    @abstractmethod
    def get_current_time_ms(self) -> int:
        """Get current time in milliseconds since epoch."""
        pass

    @abstractmethod
    def get_epoch(self) -> int:
        """Get the epoch offset."""
        pass
```

### AsyncPgRepository

```python
# src/generator/infrastructure/adapters/async_pg_repository.py
import asyncpg
from typing import Optional
from ...domain.models.snowflake_id import SnowflakeId
from ...domain.ports.persistence import IIdPersistence

class AsyncPgRepository(IIdPersistence):
    def __init__(self, dsn: str):
        self._dsn = dsn
        self._pool: Optional[asyncpg.Pool] = None

    async def connect(self):
        self._pool = await asyncpg.create_pool(self._dsn, min_size=2, max_size=10)

    async def persist(self, id: SnowflakeId, metadata: Optional[dict] = None):
        async with self._pool.acquire() as conn:
            await conn.execute("""
                INSERT INTO generated_ids
                (id, timestamp_ms, datacenter_id, machine_id, sequence_num, metadata)
                VALUES ($1, $2, $3, $4, $5, $6)
                ON CONFLICT (id) DO NOTHING
            """, id.value, id.timestamp_ms, id.datacenter_id,
               id.machine_id, id.sequence, metadata)

    async def close(self):
        await self._pool.close()
```

## Dependency Rule

```
Domain ──────► Application ──────► Infrastructure
   ▲              │                    │
   │              │                    │
   └──────────────┴────────────────────┘
              (Ports/Interfaces)

Domain has NO dependencies on Infrastructure.
Infrastructure implements Domain interfaces (ports).
```

## Dependency Injection

```python
# src/generator/main.py
from fastapi import FastAPI
from .domain.services.id_generator import IdGeneratorService
from .domain.ports.clock import IClock
from .infrastructure.adapters.system_clock import SystemClock
from .infrastructure.adapters.async_pg_repository import AsyncPgRepository

def create_app() -> FastAPI:
    clock: IClock = SystemClock()
    repository = await AsyncPgRepository(dsn="postgresql://...").connect()

    generator_service = IdGeneratorService(
        datacenter_id=1,
        machine_id=1,
        clock=clock,
        repository=repository,
    )

    app = FastAPI()
    app.state.generator = generator_service
    return app
```
